Chapter 1. Web site와 Web Page

HTML Page를 직접 만들어 본다.
HTML, CSS, JavaScript의 개념이해
Ai Chat을 활용하여 계산기 Web Page 만들기

-1. Web Site란?

인터넷에 Contents가 게시 되어 있는 장소 <(접근방식)- Web Brower

server와 client가 서로 요청(request) 하고 응답(response)하며 동작한다.

// Server : Web site의 내용 보관 <(client가 요청)- 파일 전송
// Client : Web site의 파일 요청 <(Code를 전송받음)- brower에서 실행

정리 : Web Brower의 주소창에 URL(Uniform Resource Locator)을 입력 -> sever가 file(code) 전송
-> web Brower가 실행

//위 (정리) 내용을 사진으로 표현//

-2. web page의 언어 HTML

Browser가 읽는 파일의 형식이 HTML이다.
web page는 모두 HTML code로 만들어진다. (web site에서 HTML code를 직접 확인해보자)

Vscode에서 /index.html/ 파일을 만들고 아래 내용을 작성하고 web browser에서 열어 실행해보자

/"html"

<!DOCTYPE html>
<html>
    <head>
        <title>내 첫 페이지</title>
    </head>
    <body>
        <h1>안녕하세요</h1>
        <p>이게 web page입니다.</p>
    </body>
</html>

/"html"

-> Brower -> 파일 일기 -> 화면에 그려준다. <우리 모든 web page가 이런 형식이다.>

-3. Web page의 Desigh, CSS(Cascading style sheets)

HTML만으로 디자인 하기 힘들떄 : CSS를 사용해서 (Color, background, layout)을 자세하게 설정

-각 요소의 색상, 배경색, 버튼모양, 레이아웃 정렬 등을 설정한다.
-'<style>' tage를 이용하여 작성한다.

            <button type="button" class="btn-number">.</button>
            <button type="button" class="btn-operator">=</button>
        </div>
    </div>

    <script>
        // 변수 및 DOM 요소 초기화
        const display = document.getElementById('display');
        const buttons = document.querySelectorAll('button');
        
        let currentInput = '0';  // 현재 화면에 표시되는 문자열
        let previousInput = ''; // 이전 연산 단계의 값
        let operator = null;     // 선택된 연산자 (+, -, *, /)
        let shouldResetDisplay = false; // 연산자 클릭 후 새로운 숫자를 입력받을지 여부

        // 모든 버튼에 클릭 이벤트 추가
        buttons.forEach(button => {
            button.addEventListener('click', () => {
                const buttonText = button.textContent;

                // 1. 숫자 또는 소수점 버튼 클릭 시
                if (button.classList.contains('btn-number')) {
                    handleNumber(buttonText);
                } 
                // 2. 연산자 버튼 클릭 시 (+, -, *, /)
                else if (button.classList.contains('btn-operator') && buttonText !== '=') {
                    handleOperator(buttonText);
                } 
                // 3. 기능 버튼 클릭 시 (C, +/-, %, =)
                else {
                    handleFunction(buttonText);
                }
                
                // 화면 업데이트
                display.value = currentInput;
            });
        });

        // 숫자 입력 처리 함수
        function handleNumber(num) {
            // 소수점 중복 입력 방지
            if (num === '.' && currentInput.includes('.')) return;

            // 새로 숫자를 입력해야 하거나 처음 초기 상태('0')일 때
            if (currentInput === '0' || shouldResetDisplay) {
                if (num === '.') {
                    currentInput = '0.';
                } else {
                    currentInput = num;
                }
                shouldResetDisplay = false;
            } else {
                currentInput += num;
            }
        }

        // 연산자 입력 처리 함수
        function handleOperator(nextOperator) {
            const value = parseFloat(currentInput);

            // 이미 연산자가 선택되어 있고 새로운 숫자가 입력되기 전이라면 연산자만 교체
            if (operator && shouldResetDisplay) {
                operator = nextOperator;
                return;
            }

            if (previousInput === '') {
                previousInput = currentInput;
            } else if (operator) {
                const result = calculate(parseFloat(previousInput), value, operator);
                currentInput = String(result);
                previousInput = String(result);
            }

            shouldResetDisplay = true;
            operator = nextOperator;
        }

        // 기능 버튼 처리 함수
        function handleFunction(func) {
            if (func === 'C') { // 초기화
                currentInput = '0';
                previousInput = '';
                operator = null;
                shouldResetDisplay = false;
            } 
            else if (func === '+/-') { // 양수/음수 전환
                currentInput = String(parseFloat(currentInput) * -1);
            } 
            else if (func === '%') { // 퍼센트 계산 (100으로 나눔)
                currentInput = String(parseFloat(currentInput) / 100);
            } 
            else if (func === '=') { // 결과 도출
                if (!operator || shouldResetDisplay) return;

                const result = calculate(parseFloat(previousInput), parseFloat(currentInput), operator);
                currentInput = String(result);
                previousInput = '';
                operator = null;
                shouldResetDisplay = true; // 계산 끝난 후 숫자 누르면 새로 시작
            }
        }

        // 실제 사칙연산 수행 함수
        function calculate(first, second, op) {
            switch (op) {
                case '+': return first + second;
                case '-': return first - second;
                case '*': return first * second;
                case '/': 
                    if (second === 0) return 'Error'; // 0으로 나눌 때 처리
                    return first / second;
                default: return second;
            }
        }
    </script>
</body>
</html>
